docker-compose.yml
# version: '3.8'

services:
  # 1. 后端服务 (FastAPI + Prefect 逻辑)
  backend:
    build:
      context: .
      dockerfile: docker/backend/Dockerfile
    ports:
      - "8000:8000"
    environment:
      - DATABASE_URL=sqlite:///./storage/news.db
      - CHROMA_PERSIST_DIR=./storage/chroma
      - PREFECT_API_URL=http://prefect-server:4200/api # 如果用了 docker 化的 prefect
    volumes:
      # 【核心】将宿主机的 ./storage 目录映射到容器内的 /app/storage
      # 这样 SQLite 文件和 Chroma 向量数据会保存在你的 Windows/WSL 硬盘上，容器销毁数据不丢！
      - ./storage:/app/storage
    env_file:
      - .env
    networks:
      - ai-network

  # 2. 前端服务 (Next.js)
  frontend:
    build:
      context: ./frontend
      dockerfile: ../docker/frontend/Dockerfile
    ports:
      - "3000:3000"
    environment:
      - NEXT_PUBLIC_API_URL=http://localhost:8000 # 告诉前端后端在哪
    depends_on:
      - backend
    networks:
      - ai-network

  # 3. (可选) Prefect Server UI
  # 如果你希望调度中心也容器化，取消下面注释
  prefect-server:
    image: prefecthq/prefect:3-python3.11
    command: prefect server start --host 0.0.0.0
    ports:
      - "4200:4200"
    volumes:
      - prefect_data:/root/.prefect
    networks:
      - ai-network

networks:
  ai-network:
    driver: bridge

volumes:
  prefect_data: # 如果启用了 prefect-server，取消注释Q




# 使用轻量级的 Python 官方镜像
FROM python:3.11-slim

# 设置工作目录
WORKDIR /app

# 设置环境变量，防止 Python 生成 .pyc 文件，并确保输出直接打印到终端
ENV PYTHONDONTWRITEBYTECODE=1
ENV PYTHONUNBUFFERED=1

# 安装系统级依赖 (如果 readability 或 chroma 需要编译)
RUN apt-get update && apt-get install -y --no-install-recommends \
    build-essential \
    && rm -rf /var/lib/apt/lists/*

# 复制依赖文件并安装
COPY requirements.txt .
RUN pip install --no-cache-dir -r requirements.txt

# 复制项目代码 (排除不需要的大文件，可通过 .dockerignore 控制)
COPY backend/ ./backend/
COPY pipeline/ ./pipeline/
COPY ingestion/ ./ingestion/
COPY embedding/ ./embedding/
COPY config/ ./config/

# 暴露 FastAPI 端口
EXPOSE 8000

# 默认启动 FastAPI (Prefect 任务可通过 API 触发，或在 compose 中单独定义 worker)
CMD ["uvicorn", "backend.main:app", "--host", "0.0.0.0", "--port", "8000", "--reload"]
